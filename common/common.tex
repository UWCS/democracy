% anchor link to the Executive Committee section for first appearance in group only
% basically so there's not ugly rendering hyperlinking *every* mention of it when using \exec for consistency
\newcount\execLinkLevel
\execLinkLevel=-1\relax

\ExplSyntaxOn
\cs_new:Npn \ifexecsectionTF #1#2 {
    \str_if_eq:eeTF
        { \getrefbykeydefault{sec:exec}{anchor}{} }
        { section.\theHsection }
        {#1} {#2}
}
\ExplSyntaxOff

\newcommand{\exec}{%
    \ifexecsectionTF{Executive Committee}{%
        \ifnum\execLinkLevel=\currentgrouplevel\relax
            Executive Committee%
        \else
            \execLinkLevel=\currentgrouplevel\relax
            \hyperref[sec:exec]{Executive Committee}%
        \fi
    }%
    \xspace
}
